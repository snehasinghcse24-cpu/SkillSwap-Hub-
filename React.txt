// --- HELPER COMPONENT ---
const HoverButton = ({ onClick, style, hoverStyle, children, disabled }) => {
    const [hover, setHover] = useState(false);
    return (
        <button
            style={{ ...style, ...(hover && !disabled ? hoverStyle : null), ...(disabled ? styles.buttonDisabled : null) }}
            onClick={onClick}
            onMouseEnter={() => setHover(true)}
            onMouseLeave={() => setHover(false)}
            disabled={disabled}
        >
            {children}
        </button>
    );
};

const AuthScreen = ({ onAuthSuccess }) => {
    const [isLogin, setIsLogin] = useState(true);
    const [name, setName] = useState('');
    const [email, setEmail] = useState('');
    const [password, setPassword] = useState('');
    const [isLoading, setIsLoading] = useState(false);
    const [error, setError] = useState('');
    const [validationError, setValidationError] = useState('');

    const validateForm = () => {
        if (!isLogin && !name) { setValidationError('Name is required.'); return false; }
        if (!/\S+@\S+\.\S+/.test(email)) { setValidationError('Please enter a valid email address.'); return false; }
        if (password.length < 8) { setValidationError('Password must be at least 8 characters long.'); return false; }
        setValidationError('');
        return true;
    };

    const handleAuth = useCallback(async () => {
        if (!validateForm()) return;
        setIsLoading(true);
        setError('');
        const response = isLogin
            ? await apiClient.login(email, password)
            : await apiClient.register({ name, email, password });
        setIsLoading(false);
        if (response.success) onAuthSuccess(response.user);
        else setError(response.error || 'Something went wrong.');
    }, [isLogin, name, email, password, onAuthSuccess]);
    
    return (
        <div style={styles.authContainer}>
            <div style={styles.authBox}>
                <p style={styles.authTitle}>{isLogin ? 'Welcome Back' : 'Join SkillSwapHub'}</p>
                {validationError && <p style={styles.errorText}>{validationError}</p>}
                {error && <p style={styles.errorText}>{error}</p>}
                {!isLogin && <input placeholder="Full Name" style={styles.input} value={name} onChange={(e) => setName(e.target.value)} />}
                <input placeholder="Email" style={styles.input} value={email} onChange={(e) => setEmail(e.target.value)} type="email" />
                <input placeholder="Password" style={styles.input} value={password} onChange={(e) => setPassword(e.target.value)} type="password" />
                <HoverButton style={styles.buttonPrimary} hoverStyle={styles.buttonPrimaryHover} onClick={handleAuth} disabled={isLoading}>
                    <span style={styles.buttonPrimaryText}>{isLoading ? 'Loading...' : (isLogin ? 'Login' : 'Create Account')}</span>
                </HoverButton>
                <button onClick={() => setIsLogin(!isLogin)} style={styles.authSwitchButton}>
                    <p style={styles.authSwitchText}>
                        {isLogin ? "Don't have an account? " : "Already have an account? "}
                        <span style={{ fontWeight: 'bold', color: styles.colors.primary }}>{isLogin ? 'Register' : 'Login'}</span>
                    </p>
                </button>
            </div>
        </div>
    );
};

const ChatScreen = ({ partner, onBack, onVideoCall }) => {
    const [messages, setMessages] = useState(mockChatMessages);
    const [newMessage, setNewMessage] = useState('');

    const handleSendMessage = () => {
        if (!newMessage.trim()) return;
        const optimisticMessage = { id: Date.now(), sender: 'user_alex', text: newMessage, status: 'sending'};
        setMessages(prev => [...prev, optimisticMessage]);
        setNewMessage('');
        setTimeout(() => setMessages(prev => prev.map(msg => msg.id === optimisticMessage.id ? { ...msg, status: 'sent' } : msg)), 800);
    };

    return (
        <div style={styles.chatScreen}>
            <div style={styles.chatHeader}>
                <button onClick={onBack} style={styles.chatBackButton}><span style={{ color: styles.colors.primary, fontSize: 16 }}>← Back</span></button>
                <p style={styles.chatHeaderText}>{partner.name}</p>
                <HoverButton onClick={onVideoCall} style={styles.videoCallButton} hoverStyle={styles.videoCallButtonHover}><span style={{fontSize: 20}}>📹</span></HoverButton>
            </div>
            <div style={styles.chatMessagesContainer}>
                {messages.map(msg => (<div key={msg.id} style={{... (msg.sender === 'user_alex' ? styles.messageSent : styles.messageReceived), opacity: msg.status === 'sending' ? 0.6 : 1}}><p style={{margin: 0}}>{msg.text}</p></div>))}
            </div>
             <div style={styles.chatInputContainer}>
                <input type="text" placeholder="Type a message..." style={styles.chatInput} value={newMessage} onChange={e => setNewMessage(e.target.value)} onKeyPress={e => e.key === 'Enter' && handleSendMessage()} />
                <HoverButton style={styles.sendButton} hoverStyle={styles.sendButtonHover} onClick={handleSendMessage}><span style={{fontSize: 16}}>➤</span></HoverButton>
            </div>
        </div>
    );
};

// --- NAVIGATION & MAIN APP ---
const MainTabs = ({ user, onLogout }) => {
    const [activeTab, setActiveTab] = useState('dashboard');
    const [activeChat, setActiveChat] = useState(null);
    const [inVideoCall, setInVideoCall] = useState(false);

    const handleChat = useCallback((partner) => setActiveChat(partner), []);
    const handleBack = useCallback(() => setActiveChat(null), []);
    const handleStartVideo = useCallback(() => setInVideoCall(true), []);
    const handleEndVideo = useCallback(() => setInVideoCall(false), []);

    if (inVideoCall && activeChat) return <VideoCallScreen partner={activeChat} onEndCall={handleEndVideo} />;
    if (activeChat) return <ChatScreen partner={activeChat} onBack={handleBack} onVideoCall={handleStartVideo}/>;

    return (
        <div style={styles.appContainer}>
            <div style={styles.contentContainer}>
                {activeTab === 'dashboard' && <DashboardScreen onChat={handleChat} />}
                {activeTab === 'communities' && <CommunitiesScreen />}
                {activeTab === 'profile' && <ProfileScreen user={user} onLogout={onLogout} />}
            </div>
            <div style={styles.tabBar}>
                <button style={styles.tabItem} onClick={() => setActiveTab('dashboard')}><span style={activeTab === 'dashboard' ? styles.tabTextActive : styles.tabText}>Dashboard</span></button>
                <button style={styles.tabItem} onClick={() => setActiveTab('communities')}><span style={activeTab === 'communities' ? styles.tabTextActive : styles.tabText}>Communities</span></button>
                <button style={styles.tabItem} onClick={() => setActiveTab('profile')}><span style={activeTab === 'profile' ? styles.tabTextActive : styles.tabText}>Profile</span></button>
            </div>
        </div>
    );
}

export default function App() {
    const [user, setUser] = useState(null); 
    if (!user) return (<div style={{ height: '100vh', backgroundColor: styles.colors.bgLight, fontFamily: styles.fontFamily }}><AuthScreen onAuthSuccess={setUser} /></div>);
    return (<div style={{ height: '100vh', backgroundColor: styles.colors.bgWhite, fontFamily: styles.fontFamily }}><MainTabs user={user} onLogout={() => setUser(null)} /></div>);
}
