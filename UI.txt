// --- HELPER COMPONENT FOR HOVER EFFECT ---
const HoverButton = ({ onClick, style, hoverStyle, children }) => {
    const [hover, setHover] = useState(false);
    return (
        <button
            style={{ ...style, ...(hover ? hoverStyle : null) }}
            onClick={onClick}
            onMouseEnter={() => setHover(true)}
            onMouseLeave={() => setHover(false)}
        >
            {children}
        </button>
    );
};


// --- SCREEN COMPONENTS (Web-compatible) ---

const AuthScreen = ({ onAuthSuccess }) => {
    const [isLogin, setIsLogin] = useState(true);
    const [name, setName] = useState('');
    const [email, setEmail] = useState('');
    const [password, setPassword] = useState('');

    const handleAuth = async () => {
        if (isLogin) {
            const response = await apiClient.login(email, password);
            if (response.success) onAuthSuccess(response.user);
        } else {
            const response = await apiClient.register({ name, email, password });
            if (response.success) onAuthSuccess(response.user);
        }
    };
    
    return (
        <div style={styles.authContainer}>
            <div style={styles.authBox}>
                <p style={styles.authTitle}>{isLogin ? 'Welcome Back' : 'Join SkillSwapHub'}</p>
                
                {!isLogin && <input placeholder="Full Name" style={styles.input} value={name} onChange={(e) => setName(e.target.value)} />}
                <input placeholder="Email" style={styles.input} value={email} onChange={(e) => setEmail(e.target.value)} type="email" />
                <input placeholder="Password" style={styles.input} value={password} onChange={(e) => setPassword(e.target.value)} type="password" />
                
                <HoverButton style={styles.buttonPrimary} hoverStyle={styles.buttonPrimaryHover} onClick={handleAuth}>
                    <span style={styles.buttonPrimaryText}>{isLogin ? 'Login' : 'Create Account'}</span>
                </HoverButton>

                <button onClick={() => setIsLogin(!isLogin)} style={styles.authSwitchButton}>
                    <p style={styles.authSwitchText}>
                        {isLogin ? "Don't have an account? " : "Already have an account? "}
                        <span style={{ fontWeight: 'bold', color: styles.colors.primary }}>{isLogin ? 'Register' : 'Login'}</span>
                    </p>
                </button>
            </div>
        </div>
    );
};

// --- NAVIGATION & MAIN APP ---

const MainTabs = ({ user, onLogout }) => {
    const [activeTab, setActiveTab] = useState('dashboard');
    const [activeChat, setActiveChat] = useState(null);
    const [inVideoCall, setInVideoCall] = useState(false);

    const handleChat = (partner) => setActiveChat(partner);
    const handleBack = () => setActiveChat(null);
    const handleStartVideo = () => setInVideoCall(true);
    const handleEndVideo = () => setInVideoCall(false);

    if (inVideoCall && activeChat) {
        return <VideoCallScreen partner={activeChat} onEndCall={handleEndVideo} />;
    }
    
    if (activeChat) {
        return <ChatScreen partner={activeChat} onBack={handleBack} onVideoCall={handleStartVideo}/>;
    }

    return (
        <div style={styles.appContainer}>
            <div style={styles.contentContainer}>
                {activeTab === 'dashboard' && <DashboardScreen onChat={handleChat} />}
                {activeTab === 'profile' && <ProfileScreen user={user} />}
            </div>
            <div style={styles.tabBar}>
                <button style={styles.tabItem} onClick={() => setActiveTab('dashboard')}>
                    <span style={activeTab === 'dashboard' ? styles.tabTextActive : styles.tabText}>Dashboard</span>
                </button>
                 <button style={styles.tabItem} onClick={() => setActiveTab('profile')}>
                    <span style={activeTab === 'profile' ? styles.tabTextActive : styles.tabText}>Profile</span>
                </button>
                <button style={styles.tabItem} onClick={onLogout}>
                    <span style={styles.tabText}>Logout</span>
                </button>
            </div>
        </div>
    );
}

export default function App() {
    const [user, setUser] = useState(null); 

    if (!user) {
        return (
            <div style={{ height: '100vh', backgroundColor: styles.colors.bgLight, fontFamily: styles.fontFamily }}>
                <AuthScreen onAuthSuccess={setUser} />
            </div>
        );
    }
    
    return (
        <div style={{ height: '100vh', backgroundColor: styles.colors.bgWhite, fontFamily: styles.fontFamily }}>
            <MainTabs user={user} onLogout={() => setUser(null)} />
        </div>
    );
}
