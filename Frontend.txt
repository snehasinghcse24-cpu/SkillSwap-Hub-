// --- MAIN APP CONTROLLER ---

export default function App() {
    const [page, setPage] = useState('landing'); // landing, login, register, dashboard, profile
    const [isLoggedIn, setIsLoggedIn] = useState(false);
    const [activeChat, setActiveChat] = useState(null); // { id, name, avatar }
    const [activeVideoCall, setActiveVideoCall] = useState(null);

    const handleNavigate = (targetPage) => {
        setActiveChat(null);
        setActiveVideoCall(null);
        setPage(targetPage);
    };

    const handleLogin = () => {
        setIsLoggedIn(true);
        setPage('dashboard');
    };
    
    const handleLogout = () => {
        setIsLoggedIn(false);
        setPage('landing');
    };

    const handleStartChat = (partner) => setActiveChat(partner);
    const handleStartVideoCall = (partner) => setActiveVideoCall(partner);
    
    const handleBackToApp = () => {
        setActiveChat(null);
        setActiveVideoCall(null);
    }

    if (!isLoggedIn) {
        switch (page) {
            case 'login': return <LoginPage onLogin={handleLogin} onNavigate={handleNavigate} />;
            case 'register': return <RegisterPage onLogin={handleLogin} onNavigate={handleNavigate} />;
            case 'landing':
            default:
                return <LandingPage onNavigate={handleNavigate} />;
        }
    }

    const renderMainContent = () => {
        if (activeChat) return <ChatPage partner={activeChat} onBack={handleBackToApp} />;
        if (activeVideoCall) return <VideoCallPage partner={activeVideoCall} onBack={handleBackToApp} />;
        
        switch (page) {
            case 'profile': return <ProfilePage />;
            case 'dashboard':
            default:
                return <DashboardPage onChat={handleStartChat} onVideoCall={handleStartVideoCall} />;
        }
    };

    return (
        <AppLayout onNavigate={handleNavigate} onLogout={handleLogout}>
            {renderMainContent()}
        </AppLayout>
    );
}
